\begin{ejercicio}[30 pts.]
Calcule el producto $A \cdot B$, dado que:
\[
A = \begin{pmatrix} 1 & 0 & 2 \\ -1 & 3 & 1 \end{pmatrix}, \qquad
B = \begin{pmatrix} 3 & 1 \\ 2 & 1 \\ 1 & 0 \end{pmatrix}
\]
Antes de calcular, indique las dimensiones de $A$, $B$ y del resultado $A\cdot B$.
\end{ejercicio}

\muestraSolucion{%
$A$ es $2\times 3$, $B$ es $3\times 2$ $\Rightarrow$ $A\cdot B$ es $2\times 2$.
\[
A\cdot B = \begin{pmatrix}
1(3)+0(2)+2(1) & 1(1)+0(1)+2(0)\\
(-1)(3)+3(2)+1(1) & (-1)(1)+3(1)+1(0)
\end{pmatrix}
= \begin{pmatrix} 5 & 1 \\ 4 & 2 \end{pmatrix}
\]
}

\begin{ejercicio}[30 pts.]
Calcule el producto $A \cdot B$, dado que:
\[
A = \begin{pmatrix} 2 & 4 \\ 1 & -3 \\ 0 & 5 \end{pmatrix}, \qquad
B = \begin{pmatrix} 3 & 1 & 2 \\ 4 & -1 & 0 \end{pmatrix}
\]
Antes de calcular, indique las dimensiones de $A$, $B$ y del resultado $A\cdot B$.
\end{ejercicio}

\muestraSolucion{%
$A$ es $3\times 2$, $B$ es $2\times 3$ $\Rightarrow$ $A\cdot B$ es $3\times 3$.
\[
A\cdot B = \begin{pmatrix}
2(3)+4(4) & 2(1)+4(-1) & 2(2)+4(0)\\
1(3)+(-3)(4) & 1(1)+(-3)(-1) & 1(2)+(-3)(0)\\
0(3)+5(4) & 0(1)+5(-1) & 0(2)+5(0)
\end{pmatrix}
= \begin{pmatrix} 22 & -2 & 4 \\ -9 & 4 & 2 \\ 20 & -5 & 0 \end{pmatrix}
\]
}

\begin{ejercicio}[40 pts.]
Resuelva el siguiente sistema usando el \textbf{método de Gauss} (eliminación gaussiana).
Muestre la matriz aumentada y cada operación elemental de fila realizada.
\[
\begin{cases}
x + y + z = 6\\
2x - y + z = 6\\
x + y + 2z = 9
\end{cases}
\]
\end{ejercicio}

\muestraSolucion{%
\[
\left[\begin{array}{ccc|c}1&1&1&6\\2&-1&1&6\\1&1&2&9\end{array}\right]
\xrightarrow{\substack{R_2\to R_2-2R_1\\R_3\to R_3-R_1}}
\left[\begin{array}{ccc|c}1&1&1&6\\0&-3&-1&-6\\0&0&1&3\end{array}\right]
\]
$z=3$;\quad $-3y-3=-6\Rightarrow y=1$;\quad $x=6-1-3=2$.
\[\boxed{x=2,\quad y=1,\quad z=3}\]
}
