\begin{ejercicio}[30 pts.]
Calcule el producto $A \cdot B$, dado que:
\[
A = \begin{pmatrix} 4 & -3 \\ 2 & 5 \end{pmatrix}, \qquad
B = \begin{pmatrix} 7 & 1 \\ 0 & -2 \end{pmatrix}
\]
Antes de calcular, indique las dimensiones de $A$, $B$ y del resultado $A\cdot B$.
\end{ejercicio}

\muestraSolucion{%
$A$ y $B$ son $2\times 2$ $\Rightarrow$ $A\cdot B$ es $2\times 2$.
\[
A\cdot B = \begin{pmatrix}
4(7)+(-3)(0) & 4(1)+(-3)(-2)\\
2(7)+5(0) & 2(1)+5(-2)
\end{pmatrix}
= \begin{pmatrix} 28 & 10 \\ 14 & -8 \end{pmatrix}
\]
}

\begin{ejercicio}[30 pts.]
Calcule el producto $A \cdot B$, dado que:
\[
A = \begin{pmatrix} 3 & 2 \\ 1 & 0 \\ -2 & 5 \end{pmatrix}, \qquad
B = \begin{pmatrix} 1 & -1 & 2 \\ 3 & 0 & 4 \end{pmatrix}
\]
Antes de calcular, indique las dimensiones de $A$, $B$ y del resultado $A\cdot B$.
\end{ejercicio}

\muestraSolucion{%
$A$ es $3\times 2$, $B$ es $2\times 3$ $\Rightarrow$ $A\cdot B$ es $3\times 3$.
\[
A\cdot B = \begin{pmatrix}
3(1)+2(3) & 3(-1)+2(0) & 3(2)+2(4)\\
1(1)+0(3) & 1(-1)+0(0) & 1(2)+0(4)\\
(-2)(1)+5(3) & (-2)(-1)+5(0) & (-2)(2)+5(4)
\end{pmatrix}
= \begin{pmatrix} 9 & -3 & 14 \\ 1 & -1 & 2 \\ 13 & 2 & 16 \end{pmatrix}
\]
}

\begin{ejercicio}[40 pts.]
Resuelva el siguiente sistema usando el \textbf{método de Gauss} (eliminación gaussiana).
Muestre la matriz aumentada y cada operación elemental de fila realizada.
\[
\begin{cases}
x + y + z = 10\\
x + 2y - z = 7\\
2x - y + z = 8
\end{cases}
\]
\end{ejercicio}

\muestraSolucion{%
\[
\left[\begin{array}{ccc|c}1&1&1&10\\1&2&-1&7\\2&-1&1&8\end{array}\right]
\xrightarrow{\substack{R_2\to R_2-R_1\\R_3\to R_3-2R_1}}
\left[\begin{array}{ccc|c}1&1&1&10\\0&1&-2&-3\\0&-3&-1&-12\end{array}\right]
\xrightarrow{R_3\to R_3+3R_2}
\left[\begin{array}{ccc|c}1&1&1&10\\0&1&-2&-3\\0&0&-7&-21\end{array}\right]
\]
$z=3$;\quad $y-6=-3\Rightarrow y=3$;\quad $x=10-3-3=4$.
\[\boxed{x=4,\quad y=3,\quad z=3}\]
}
